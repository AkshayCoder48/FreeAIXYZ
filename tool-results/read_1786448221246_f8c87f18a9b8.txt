    37→export interface ModelCapabilities {
    38→  /** Returns token-by-token SSE deltas (true upstream streaming). */
    39→  streaming: boolean;
    40→  /** OpenAI-style function/tool calling via prompt injection. */
    41→  tools: boolean;
    42→  /** Accepts a system prompt. */
    43→  systemPrompt: boolean;
    44→  /** Multi-turn conversation history. */
    45→  multiTurn: boolean;
    46→  /** Image / vision inputs. */
    47→  vision: boolean;
    48→  /** Live web search for grounded, up-to-date answers. */
    49→  webSearch: boolean;
    50→}
    51→
    52→export interface GatewayModel {
    53→  id: string;
    54→  provider: ProviderId;
    55→  /** Upstream model id sent to the provider. */
    56→  upstream: string;
    57→  description: string;
    58→  /** Short label for chips/badges. */
    59→  category: "professional" | "sfw" | "unrestricted" | "reasoning";
    60→  capabilities: ModelCapabilities;
    61→  /** Max context window (approx, in tokens). 0 = unknown. */
    62→  contextWindow: number;
    63→  /** Whether the model is currently reachable from this gateway. */
    64→  experimental?: boolean;
    65→  /** Modality — text models answer prompts; text-to-image models emit images. */
    66→  modality?: "text" | "text-to-image";
    67→  /** For text-to-image models only: visual style family. */
    68→  imageCategory?: "anime" | "realism" | "unrestricted-anime" | "unrestricted-realism" | "mixed" | "general";
    69→}

   637→/** Find a model by id (case-insensitive). Returns undefined if not found. */
   638→export function findModel(id: string | undefined): GatewayModel | undefined {
   639→  if (!id) return undefined;
   640→  const lower = id.toLowerCase();
   641→  return MODELS.find((m) => m.id.toLowerCase() === lower);
   642→}

   647→/** Resolve a requested model id. Falls back to the default if unknown. */
   648→export function resolveGatewayModel(
