   208→interface CodeBlockExtract {
   209→  language: string;
   210→  code: string;
   211→}

   213→/** Extract fenced code blocks (```lang\ncode```) from a markdown-ish string. */
   214→function extractCodeBlocks(content: string): CodeBlockExtract[] {
   215→  const blocks: CodeBlockExtract[] = [];
   216→  const regex = /```([\w+-]*)\n([\s\S]*?)```/g;
   217→  let match: RegExpExecArray | null;
   218→  while ((match = regex.exec(content)) !== null) {
   219→    const language = (match[1] || "text").toLowerCase();
   220→    const code = match[2].replace(/\n$/, "");
   221→    blocks.push({ language, code });
   222→  }
   223→  return blocks;
   224→}

   226→/** Accumulate streaming tool_call deltas by index (OpenAI streaming format). */
   227→function accumulateToolCalls(
   228→  existing: ToolCall[] | undefined,
   229→  deltas: Array<{
   230→    index?: number;
   231→    id?: string;
   232→    function?: { name?: string; arguments?: string };
   233→  }>,
   234→): ToolCall[] {
   235→  const acc: ToolCall[] = existing
   236→    ? existing.map((t) => ({
   237→        id: t.id,
   238→        function: {
   239→          name: t.function.name,
   240→          arguments: t.function.arguments,
   241→        },
   242→      }))
   243→    : [];
   244→  for (const d of deltas) {
   245→    const idx = d.index ?? 0;
   246→    if (!acc[idx]) {
   247→      acc[idx] = {
   248→        id: d.id ?? "",
   249→        function: {
   250→          name: d.function?.name ?? "",
   251→          arguments: d.function?.arguments ?? "",
   252→        },
   253→      };
   254→    } else {
   255→      if (d.id) acc[idx].id = d.id;
   256→      if (d.function?.name) acc[idx].function.name = d.function.name;
   257→      if (d.function?.arguments)
   258→        acc[idx].function.arguments += d.function.arguments;
   259→    }
   260→  }
   261→  return acc;
   262→}

   264→function formatTimestamp(ts: number): string {
   265→  const d = new Date(ts);
   266→  const now = new Date();
   267→  const sameDay =
   268→    d.getFullYear() === now.getFullYear() &&
   269→    d.getMonth() === now.getMonth() &&
   270→    d.getDate() === now.getDate();
   271→  const time = d.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" });
   272→  if (sameDay) return time;
   273→  return `${d.toLocaleDateString([], { month: "short", day: "numeric" })} ${time}`;
   274→}

   276→// ─── SSE Parser ─────────────────────────────────────────────────────────────
   277→
   278→interface SSEDelta {
   279→  content?: string;
   280→  toolCalls?: ToolCall[];
   281→}
   282→
   283→/** Parse an SSE stream into incremental content + tool-call deltas. */
   284→async function* parseSSE(response: Response): AsyncGenerator<SSEDelta> {
   285→  const reader = response.body?.getReader();
   286→  if (!reader) return;
   287→  const decoder = new TextDecoder();
   288→  let buffer = "";
   289→  while (true) {
   290→    const { done, value } = await reader.read();
   291→    if (done) break;
   292→    buffer += decoder.decode(value, { stream: true });
   293→    const lines = buffer.split("\n");
   294→    buffer = lines.pop() ?? "";
   295→    for (const line of lines) {
   296→      const trimmed = line.trim();
   297→      if (!trimmed.startsWith("data:")) continue;
   298→      const data = trimmed.slice(5).trim();
   299→      if (data === "[DONE]") return;
   300→      try {
   301→        const json = JSON.parse(data);
   302→        const delta = json.choices?.[0]?.delta;
   303→        if (!delta) continue;
   304→        const content =
   305→          typeof delta.content === "string" ? delta.content : undefined;
   306→        const toolCalls = Array.isArray(delta.tool_calls)
   307→          ? accumulateToolCalls([], delta.tool_calls)
   308→          : undefined;
   309→        if (content || toolCalls) {
   310→          yield { content, toolCalls };
   311→        }
   312→      } catch {
   313→        /* ignore keep-alive / partial frames */
   314→      }
   315→    }
   316→  }
   317→}

   364→function MessageContent({ content }: { content: string }) {
   365→  if (!content) return null;
   366→  // Split on fenced code blocks; keep them so we can render separately.
   367→  const parts = content.split(/(```[\w+-]*\n[\s\S]*?```)/g);
   368→  return (
   369→    <div className="space-y-1">
   370→      {parts.map((part, i) => {
   371→        if (part.startsWith("```")) {
   372→          const match = part.match(/^```([\w+-]*)\n([\s\S]*?)```$/);
   373→          if (match) {
   374→            return (
   375→              <CodeBlock
   376→                key={i}
   377→                language={match[1] || "text"}
   378→                code={match[2].replace(/\n$/, "")}
   379→              />
   380→            );
   381→          }
   382→        }
   383→        // Render plain text with preserved whitespace + basic inline code.
   384→        return part ? (
   385→          <div
   386→            key={i}
   387→            className="whitespace-pre-wrap break-words leading-relaxed"
   388→          >
   389→            {renderInlineCode(part)}
   390→          </div>
   391→        ) : null;
   392→      })}
   393→    </div>
   394→  );
   395→}
   396→
   397→/** Render `inline code` spans within plain text. */
   398→function renderInlineCode(text: string): ReactNode[] {
   399→  const nodes: ReactNode[] = [];
   400→  const regex = /`([^`\n]+)`/g;
   401→  let last = 0;
   402→  let match: RegExpExecArray | null;
   403→  let i = 0;
   404→  while ((match = regex.exec(text)) !== null) {
   405→    if (match.index > last) {
   406→      nodes.push(text.slice(last, match.index));
   407→    }
   408→    nodes.push(
   409→      <code
   410→        key={`ic-${i}`}
   411→        className="px-1.5 py-0.5 rounded bg-zinc-900/80 border border-border/60 text-[12px] font-mono text-[#ff9a3c]"
   412→      >
   413→        {match[1]}
   414→      </code>,
   415→    );
   416→    last = regex.lastIndex;
   417→    i++;
   418→  }
   419→  if (last < text.length) nodes.push(text.slice(last));
   420→  return nodes;
   421→}

   662→  // ─── Chat completions (streaming + non-streaming) ────────────────────────

   680→          `HTTP ${res.status}${errText ? `: ${errText.slice(0, 200)}` : ""}`,
   681→        );
   682→      }

   700→                  toolCalls && toolCalls.length > 0 ? toolCalls : undefined,
   701→              };
   702→            }
   703→            return next;
   704→          });
   705→        }
   706→        if (!acc && (!toolCalls || toolCalls.length === 0)) {
   707→          setMessages((prev) => {
   708→            const next = [...prev];
   709→            next[next.length - 1] = {
   710→              role: "assistant",
   711→              content: "(empty response)",
   712→            };
   713→            return next;
   714→          });
   715→        } else {
   716→          // After stream completes, extract any code blocks as Files.
   717→          if (acc) saveFilesFromContent(acc);
   718→        }
   719→      } else {
   720→        const data = await res.json();

   722→        const content: string = msg?.content ?? "(empty)";
   723→        const toolCalls: ToolCall[] | undefined = Array.isArray(
   724→          msg?.tool_calls,
   725→        )
   726→          ? msg.tool_calls
   727→          : undefined;
   728→        setMessages((prev) => {
   729→          const next = [...prev];
   730→          next[next.length - 1] = {
   731→            role: "assistant",
   732→            content,
   733→            toolCalls:
   734→              toolCalls && toolCalls.length > 0 ? toolCalls : undefined,
   735→          };
   736→          return next;
   737→        });
   738→        if (content) saveFilesFromContent(content);
   739→      }
   740→    },
   741→    [model, stream, saveFilesFromContent],
