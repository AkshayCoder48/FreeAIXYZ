   115→
   116→    const newImage: GeneratedImage = {
   117→      id: `poll-${Date.now()}-${seed}`,
   118→      url: imageUrl,
   119→      prompt,
   120→      source: "pollinations",
   121→      timestamp: Date.now(),
   122→      width,
   123→      height,
   124→    };
   125→
   126→    setCurrentImage(newImage);
   127→    setImageLoading(true);
   128→    setIsGenerating(false);
   129→
   130→    // Add to gallery after we know the image loads
   131→    // We track loading state via the img onLoad/onError
   132→  }, [prompt, width, height]);
   133→
   134→  // ── Generate with AIAnime.io (client-side fetch) ──
   135→  const generateAianime = useCallback(async () => {
   136→    try {
   137→      const formData = new URLSearchParams();
   138→      formData.append("prompt", prompt);
   139→      formData.append("model_type", "anime_io");
   140→
   141→      const response = await fetch(
   142→        "https://api.aianime.io/api/image-generate/text2image",
   143→        {
   144→          method: "POST",
   145→          headers: {
   146→            "Content-Type": "application/x-www-form-urlencoded",
   147→            Accept: "application/json",
   148→          },
   149→          body: formData.toString(),
   150→        }
   151→      );
   152→
   153→      const data = await response.json();
   154→
   155→      if (data.code === 404 || !data.result) {
   156→        toast.error("AIAnime.io returned an error. Falling back to Pollinations.ai");
   157→        // Fallback to Pollinations
   158→        setSource("pollinations");
   159→        generatePollinations();
   160→        return;
   161→      }
   162→
   163→      // The result could be a URL or base64
   164→      let imageUrl = "";
   165→      if (typeof data.result === "string") {
   166→        imageUrl = data.result;
   167→      } else if (data.result?.images?.[0]) {
   168→        imageUrl = data.result.images[0];
   169→      } else if (data.result?.url) {
   170→        imageUrl = data.result.url;
   171→      } else if (data.result?.image) {
   172→        imageUrl = data.result.image;
   173→      } else {
   174→        // Try to extract any URL-like string from result
   175→        const resultStr = JSON.stringify(data.result);
   176→        const urlMatch = resultStr.match(/https?:\/\/[^\s"']+/);
   177→        if (urlMatch) {
   178→          imageUrl = urlMatch[0];
   179→        }
   180→      }
   181→
   182→      if (!imageUrl) {
   183→        toast.error("Could not extract image from AIAnime.io response. Falling back to Pollinations.ai");
   184→        setSource("pollinations");
   185→        generatePollinations();
   186→        return;
   187→      }
   188→
   189→      const newImage: GeneratedImage = {
   190→        id: `ai-${Date.now()}`,
   191→        url: imageUrl,
   192→        prompt,
   193→        source: "aianime",
   194→        timestamp: Date.now(),
   195→        width,
   196→        height,
   197→      };
   198→
   199→      setCurrentImage(newImage);
   200→      setImageLoading(true);
   201→      setGallery((prev) => [newImage, ...prev]);
   202→      setIsGenerating(false);
   203→      toast.success("Image generated with AIAnime.io!");
   204→    } catch {
   205→      toast.error("AIAnime.io fetch failed. Falling back to Pollinations.ai");
   206→      setSource("pollinations");
   207→      generatePollinations();
   208→    }
   209→  }, [prompt, width, height, generatePollinations]);

   213→    if (!prompt.trim()) {
   214→      toast.error("Please enter a prompt first!");
   215→      return;
   216→    }
   217→
   218→    setIsGenerating(true);
   219→
   220→    if (source === "pollinations") {
   221→      generatePollinations();
   222→    } else {
   223→      generateAianime();
   224→    }
   225→  }, [prompt, source, generatePollinations, generateAianime]);
   226→

   232→      setGallery((prev) => {
   233→        if (prev.some((img) => img.id === currentImage.id)) return prev;
   234→        return [currentImage, ...prev];
   235→      });

   238→  }, [currentImage]);

   241→    setImageLoading(false);
   242→    if (currentImage?.source === "pollinations") {
   243→      toast.error("Failed to load image from Pollinations.ai. Try again with a different seed.");
   244→    } else {
   245→      toast.error("Failed to load image. Try a different source or prompt.");

   278→      }
   279→    },
   280→    []

   307→          <motion.div

   311→            className="flex items-center justify-center gap-3"

   317→              <Sparkles className="w-7 h-7 sm:w-8 sm:h-8 text-fuchsia-400" />
   318→            </motion.div>
   319→            <h1 className="text-3xl sm:text-4xl md:text-5xl font-bold gradient-text tracking-tight">
   320→              Anime AI Studio
   321→            </h1>
   322→            <motion.div
   323→              animate={{ rotate: [360, 0] }}
   324→              transition={{ duration: 8, repeat: Infinity, ease: "linear" }}
   325→            >
   326→              <Wand2 className="w-7 h-7 sm:w-8 sm:h-8 text-pink-400" />
   327→            </motion.div>
   328→          </motion.div>
   329→          <motion.p

   349→            <CardContent className="p-4 sm:p-6 space-y-5">
   350→              {/* Source Selector */}

   361→                  <TabsList className="w-full grid grid-cols-2 bg-muted/50 h-auto p-1">
   362→                    <TabsTrigger
   363→                      value="pollinations"

   380→                <p className="text-xs text-muted-foreground">
   381→                  {source === "pollinations" ? (

   383→                      <Server className="w-3 h-3" />
